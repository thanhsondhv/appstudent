# --- LOGIC XÁC THỰC MẬT KHẨU ASP.NET V3 ---
def verify_aspnet_v3(hashed_pass: str, raw_pass: str) -> bool:
    try:
        decoded = base64.b64decode(hashed_pass)
        # Chuẩn V3: Byte 0 là version(0x01), byte 13-28 là salt, byte 29-60 là hash
        salt = decoded[13:29]
        stored_key = decoded[29:61]
        generated_key = hashlib.pbkdf2_hmac('sha256', raw_pass.encode(), salt, 10000, 32)
        return hmac.compare_digest(generated_key, stored_key)
    except:
        return False

# --- 2. XỬ LÝ LOGIC ĐĂNG NHẬP ---
@app.post("/login")
def login(username: str, password: str):
    try:
        with pyodbc.connect(CONN_STR) as conn:
            cursor = conn.cursor()
            # Truy vấn PasswordHash từ DB Core qua Linked Server
            sql = f"SELECT * FROM OPENQUERY([172.16.95.200], 'SELECT PasswordHash, FullNameNew FROM DHVINH_Stagging_Core.Identityserver.Users WHERE UserName = ''{username}''')"
            cursor.execute(sql)
            row = cursor.fetchone()
            
            if not row:
                raise HTTPException(status_code=404, detail="Không tìm thấy tài khoản")
            if not row.PasswordHash:
                raise HTTPException(status_code=401, detail="Tài khoản chưa có mật khẩu (NULL)")
                
            if verify_aspnet_v3(row.PasswordHash, password):
                return {"status": "success", "user": row.FullNameNew}
            
            raise HTTPException(status_code=401, detail="Mật khẩu không chính xác")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

                SELECT * FROM OPENQUERY([172.16.95.200], 
                'SELECT sv.HoTen, sv.MaLopHP, lhp.Ten AS TenLop, tkb.MaPhong, tkb.NgayHoc, tkb.TietHoc, tkb.TuanHocFulls
