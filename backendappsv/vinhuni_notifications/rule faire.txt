rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // 1. Quy tắc cho chính cái Nhóm (Tên nhóm, danh sách thành viên)
    match /groups/{groupId} {
      // Cho phép cán bộ đọc thông tin nhóm nếu họ có tên trong mảng 'members'
      allow read: if request.auth != null && request.auth.uid in resource.data.members;
      
      // 2. Quy tắc cho Tin nahắn bên trong nhóm đó
      match /messages/{messageId} {
        // Chỉ thành viên trong nhóm mới được xem và gửi tin nhắn
        allow read, write: if request.auth != null && 
          request.auth.uid in get(/databases/$(database)/documents/groups/$(groupId)).data.members;
      }
    }

    // 3. (Tùy chọn) Quy tắc cho thông tin cá nhân cán bộ nếu cần
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
